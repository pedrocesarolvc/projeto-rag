# Dependências do projeto, organizadas pela camada da stack (ver
# docs/documentacao.md). Cada bloco entra em uso quando a etapa
# correspondente é implementada.

# API
fastapi
uvicorn[standard]
python-multipart  # necessário para o FastAPI aceitar upload de arquivo (POST /documentos)

# Validação (contratos entre etapas — app/modelos.py)
pydantic[email]

# Configuração via variável de ambiente — app/config.py
python-dotenv

# Extração de PDF — Etapa 2 (app/ingestao/extrator.py)
pymupdf

# Modelo de embedding local — Etapa 4 (app/indexacao/embedder.py)
sentence-transformers

# Banco de dados: PostgreSQL + pgvector — Etapa 4 (armazenamento)
psycopg[binary]
pgvector

# LLM — Etapa 6 (app/geracao/llm.py). Ollama para uso local; groq só
# entra em ação quando LLM_PROVIDER=groq (instância pública, ver
# app/config.py) — mas o pacote fica instalado sempre, é leve (só um
# cliente HTTP) e assim a mesma imagem serve os dois ambientes.
ollama
groq

# Autenticação — Etapa 7 (app/auth/): hash de senha e token de sessão
bcrypt
pyjwt

# Testes
pytest
httpx  # TestClient do FastAPI (testes de ponta a ponta — Etapa 7)
